import de.fraunhofer.aisec.cpg.graph.edges.flows.ContextSensitiveDataflow
import de.fraunhofer.aisec.cpg.graph.edges.flows.FieldDataflowGranularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.FullDataflowGranularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.Granularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.IndexedDataflowGranularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.PartialDataflowGranularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.PointerDataflowGranularity
import de.fraunhofer.aisec.cpg.graph.edges.flows.StringIndexedDataflowGranularity

import de.fraunhofer.aisec.cpg.passes.PointsToPass
import de.fraunhofer.aisec.cpg.passes.EvaluationOrderGraphPass
import de.fraunhofer.aisec.cpg.passes.NonLocalVariablesIdentificationPass
import de.fraunhofer.aisec.cpg.passes.ImportResolver
import de.fraunhofer.aisec.cpg.passes.SymbolResolver
import de.fraunhofer.aisec.cpg.passes.TypeHierarchyResolver
import de.fraunhofer.aisec.cpg.passes.TypeResolver
import de.fraunhofer.aisec.cpg.test.analyze
import java.io.File
import kotlin.time.Duration.Companion.seconds
import java.util.IdentityHashMap
import kotlin.test.Test

/** Scratch: canonical dump of the Kotlin CPG (frontend + the passes SymbolResolver needs). */
class TmpDump {
    /** What a partial granularity points to: the name of a node when details are wanted. */
    private fun partialDetail(t: Any?): String =
        if (t is de.fraunhofer.aisec.cpg.graph.Node) {
            if (System.getenv("DUMP_CTX") != null) "${t.javaClass.simpleName}:${t.name}" else "node"
        } else {
            "$t"
        }

    private fun granularity(g: Granularity): String =
        when (g) {
            is FullDataflowGranularity -> ""
            is PointerDataflowGranularity -> ":P(${g.pointerTarget})"
            is FieldDataflowGranularity -> ":F(${g.partialTarget.name})"
            is IndexedDataflowGranularity -> ":I(${g.partialTarget})"
            is StringIndexedDataflowGranularity -> ":S(${g.partialTarget})"
            is PartialDataflowGranularity<*> ->
                ":Part(${partialDetail(g.partialTarget)})"
            else -> ":?"
        }

    @Test
    fun dump() {
        val dir = File(System.getenv("DUMP_DIR") ?: error("DUMP_DIR not set"))
        val outDir = File(System.getenv("DUMP_OUT_DIR") ?: error("DUMP_OUT_DIR not set"))
        outDir.mkdirs()
        dir.listFiles { f -> f.isFile && f.extension == "c" }!!
            .sortedBy { it.name }
            .forEach { file ->
                val out = File(outDir, file.name + ".kt.txt")
                try {
                    out.writeText(dumpFile(file))
                } catch (e: Throwable) {
                    out.writeText("ERROR ${e.javaClass.simpleName}: ${e.message}\n")
                }
            }
    }

    private fun dumpFile(file: File): String {
        val result =
            analyze(listOf(file), file.parentFile.toPath(), false) {
                it.registerLanguage<CLanguage>()
                it.loadIncludes(false)
                it.debugParser(false)
                it.inferenceConfiguration(
                    de.fraunhofer.aisec.cpg.InferenceConfiguration.builder().enabled(false).build()
                )
                it.registerPass<ImportResolver>()
                it.registerPass<TypeResolver>()
                it.registerPass<TypeHierarchyResolver>()
                it.registerPass<EvaluationOrderGraphPass>()
                it.registerPass<SymbolResolver>()
                // DUMP_PASSES: dfg adds the data flow graph (without control flow)
                val wanted = (System.getenv("DUMP_PASSES") ?: "").split(",")
                if ("dfg" in wanted || "pointsto" in wanted) {
                    it.registerPass<NonLocalVariablesIdentificationPass>()
                    it.registerPass<DFGPass>()
                }
                // pointsto: the PointsToPass is part of the pipeline (references get their flow
                // sensitive edges from it)
                if ("pointsto" in wanted) {
                    it.registerPass<PointsToPass>()
                    it.configurePass<PointsToPass>(PointsToPass.Configuration(timeout = 30.seconds))
                }
            }
        val tus = result.components.flatMap { it.translationUnits }

        val ids = IdentityHashMap<Any, Int>()
        fun number(n: AstNode) {
            ids[n] = ids.size
            n.astChildren.forEach { number(it) }
        }
        tus.forEach { number(it) }

        fun ref(d: Any?) =
            if (d == null) "-"
            else
                ids[d]?.let { "#$it" }
                    ?: ("ext:${d.javaClass.simpleName}" +
                        (if (System.getenv("DUMP_CTX") != null && d is de.fraunhofer.aisec.cpg.graph.Node)
                            "(${d.name})"
                        else ""))
        val sb = StringBuilder()
        fun dump(n: AstNode, depth: Int) {
            val r = n.location?.region
            val loc =
                if (r == null) "?"
                else "${r.startLine}:${r.startColumn}-${r.endLine}:${r.endColumn}"
            val type = (n as? HasType)?.type?.name?.toString() ?: "-"
            val code = (n.code ?: "").replace("\n", "\\n").take(60)
            sb.append("  ".repeat(depth))
                .append(
                    "#${ids[n]} ${n.javaClass.simpleName} name='${n.name}' type='$type' loc=$loc code='$code'"
                )
            if (System.getenv("DUMP_ACCESS") != null && n is Expression && n.access != AccessValues.READ) {
                sb.append(" acc=${n.access}")
            }
            if (n is Reference) sb.append(" refersTo=${ref(n.refersTo)}")
            if (n is Call) sb.append(" invokes=[${n.invokes.joinToString(",") { ref(it) }}]")
            val eog =
                n.nextEOGEdges.joinToString(",") {
                    ref(it.end) +
                        (it.branch?.let { b -> if (b) ":T" else ":F" } ?: "") +
                        (if (it.unreachable) ":U" else "")
                }
            if (eog.isNotEmpty()) sb.append(" eog=[$eog]")
            val dfg =
                n.prevDFGEdges.joinToString(",") { e ->
                    ref(e.start) + granularity(e.granularity) +
                        (if (e.functionSummary) "!fs" else "") +
                        (if (System.getenv("DUMP_CTX") != null && e is ContextSensitiveDataflow)
                            e.callingContexts.joinToString("") { c ->
                                (if (c is CallingContextIn) "<in:" else if (c is CallingContextOut) "<out:" else "<?:") +
                                    c.calls.joinToString("/") { call -> ref(call) } + ">"
                            }
                        else "")
                }
            if (dfg.isNotEmpty()) sb.append(" dfg=[$dfg]")
            sb.append('\n')
            n.astChildren.forEach { dump(it, depth + 1) }
        }
        tus.forEach { dump(it, 0) }
        return sb.toString()
    }
}
